jsonx

A tiny htmx-style framework for JSON APIs. HTML attributes call your API, EJS templates turn the JSON into HTML in the browser, and it plays nicely with web components.

<button jx-get="/api/todos" jx-template="#todo-list" jx-target="#todos">Load</button>
<ul id="todos"></ul>

<script type="text/ejs" id="todo-list">
  <% data.forEach(function (todo) { %>
    <li><%= todo.title %></li>
  <% }) %>
</script>
