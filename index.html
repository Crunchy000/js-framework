jsonx

A tiny htmx-style framework for JSON APIs. Your server sends JSON shaped for the page, a plain HTML template draws it in the browser, and only what changed is updated. It plays nicely with web components.

<wa-card jx-get="api/todos">
  <template>
    <ul>
      <jx-each items="todos">
        <li class:done="{{done}}">
          <wa-checkbox checked="{{done}}" jx-post="api/todos/{{id}}/toggle">{{title}}</wa-checkbox>
        </li>
      </jx-each>
    </ul>
    <jx-if test="!todos">
      <p>Nothing to do.</p>
    </jx-if>
  </template>
</wa-card>
